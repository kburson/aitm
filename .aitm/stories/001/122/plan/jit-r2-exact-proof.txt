class _ServiceEvidence:
    """Stable source-only references independent of destination project keys."""

    key: bytes = field(repr=False)

    @property
    def key_id(self) -> str:
        return sha256(self.key).hexdigest()

    def as_of(self) -> LedgerView:
        namespace = UUID(
            bytes=hmac.digest(self.key, b"monitor-source-namespace/v1", "sha256")[:16]
        )
        return LedgerView(namespace, 0, (), ())

    def _tag(self, digest: str) -> str:
        return hmac.digest(
            self.key, b"monitor-source-proof/v1\0" + bytes.fromhex(digest), "sha256"
        ).hex()

    def pseudonymize(self, namespace: str, raw_reference: str) -> SourceRef:
        if not re.fullmatch(r"[a-z][a-z0-9-]{0,63}", namespace) or not raw_reference:
            raise ValueError("Expected a monitor source namespace and identifier")
        message = f"monitor-source-ref/v1\0{namespace}\0{raw_reference}".encode()
        digest = hmac.digest(self.key, message, "sha256").hex()
        return SourceRef(f"hmac-sha256:{digest}:{self._tag(digest)}")

    def validate_source_ref(self, reference: SourceRef) -> None:
        if type(reference) is not SourceRef:
            raise LedgerError("Unverified monitor source reference")
        _, digest, tag = reference.value.split(":")
        if not hmac.compare_digest(tag, self._tag(digest)):
            raise LedgerError("Unverified monitor source reference")

def _pin_directory(path: Path) -> Iterator[_PinnedPath]:
    identities = _validated_path(path, directory=True)

def _physical_ancestry(
    path: Path,
) -> tuple[tuple[tuple[int, int], tuple[str, ...]], ...]:
    # Each existing directory anchors the remaining path, including a missing tail.
    components = tuple(
        unicodedata.normalize("NFC", part).casefold() for part in path.parts
    )
    return tuple(
        (identity, components[depth:])
        for depth, identity in enumerate(
            _validated_path(path, directory=True, allow_missing=True), start=1
        )
    )


def _guard_path(root: Path, project_id: UUID) -> Path:

def _publish(
    directory: _PinnedPath,
    name: str,
    data: bytes,
    *,
    replace_pending: bool = False,
    before_commit: Callable[[], None] | None = None,
) -> bool:
    """Publish fsynced bytes relative to a pinned private directory."""
    directory.check()
    temporary = f".publishing-{secrets.token_hex(16)}"
    fd = os.open(
        temporary,
        os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW,
        0o600,
        dir_fd=directory.fd,
    )
    try:
        with os.fdopen(fd, "wb") as handle:
            os.fchmod(fd, 0o600)
            _private(os.fstat(fd))
            _check_acl(fd)
            handle.write(data)
            handle.flush()
            os.fsync(fd)
        directory.check()
        if before_commit is not None:
            before_commit()
        if replace_pending:
            os.replace(
                temporary, name, src_dir_fd=directory.fd, dst_dir_fd=directory.fd
            )
        else:
            try:
                if not publish_noreplace(directory.fd, temporary, directory.fd, name):
                    return False
            except HostFilesystemError:
                raise LedgerError("Unable to publish private inbox file") from None
        os.fsync(directory.fd)
        directory.check()
        return True
    finally:
        try:
            os.unlink(temporary, dir_fd=directory.fd)
        except FileNotFoundError:
            pass
